cmake_minimum_required(VERSION 3.22)
set(EXTRA_COMPONENT_DIRS "../../components")
# Include tests from every application component that provides a test directory.
file(GLOB component_tests CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_LIST_DIR}/../../components/*/test/CMakeLists.txt")
set(TEST_COMPONENTS "")
foreach(test_file IN LISTS component_tests)
    get_filename_component(test_dir "${test_file}" DIRECTORY)
    get_filename_component(component_dir "${test_dir}" DIRECTORY)
    get_filename_component(component_name "${component_dir}" NAME)
    list(APPEND TEST_COMPONENTS "${component_name}")
endforeach()
if(NOT TEST_COMPONENTS)
    message(FATAL_ERROR "No application component tests were found")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(frame_codec_tests)
